Node.js backend setup for custom down payments

1. Express Node.js Create Down Payment Intent (CAD) 
const express = require('express');
const stripe = require('stripe')(process.env.STRIPE_SECRET_KEY);
const app = express();

app.post('/api/create-downpayment-intent', async (req, res) => {
  const { quoteAmountCAD, depositPercent, projectId } = req.body;
  
  // Convert down payment in CAD to cents
  const depositCents = Math.round((quoteAmountCAD * (depositPercent / 100)) * 100);

  const paymentIntent = await stripe.paymentIntents.create({
    amount: depositCents,
    currency: 'cad', // Configured strictly for Canadian Dollars
    metadata: {
      projectId: projectId,
      totalQuoteCAD: quoteAmountCAD,
      depositPercent: `${depositPercent}%`
    }
  });

  res.send({ clientSecret: paymentIntent.client_secret });
});
2. Webhook Handler to Trigger Build Work 
app.post('/api/stripe-webhook', express.raw({type: 'application/json'}), (req, res) => {
  const sig = req.headers['stripe-signature'];
  let event;

  try {
    event = stripe.webhooks.constructEvent(req.body, sig, process.env.STRIPE_WEBHOOK_SECRET);
  } catch (err) {
    return res.status(400).send(`Webhook Error: ${err.message}`);
  }

  if (event.type === 'payment_intent.succeeded') {
    const paymentIntent = event.data.object;
    // Down payment received -> trigger engineering build pipeline!
    console.log('Down payment paid for project:', paymentIntent.metadata.projectId);
  }

  res.json({ received: true });
});
